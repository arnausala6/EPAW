CREATE TABLE users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name VARCHAR(20) NOT NULL,
  password VARCHAR(20) NOT NULL,
  picture VARCHAR(255),
  role VARCHAR(20) NOT NULL DEFAULT 'user',
  username VARCHAR(30),
  email VARCHAR(255),
  age INTEGER,
  gender VARCHAR(10),
  description VARCHAR(300),
  country VARCHAR(100)
);

CREATE TABLE groups (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name VARCHAR(100) NOT NULL
);
